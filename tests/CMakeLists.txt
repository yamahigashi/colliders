cmake_minimum_required(VERSION 3.10)
project(collider_draw_tests LANGUAGES CXX)
set(CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../cmake")
find_package(Maya REQUIRED)
enable_testing()
add_executable(draw_geometry_test draw_geometry_test.cpp)
target_compile_features(draw_geometry_test PRIVATE cxx_std_11)
target_include_directories(draw_geometry_test PRIVATE ../sources "${MAYA_DEVKIT_DIR}/include")
target_compile_definitions(draw_geometry_test PRIVATE ${MAYA_COMPILE_DEFINITIONS})
foreach(library Foundation OpenMaya OpenMayaRender)
    target_link_libraries(draw_geometry_test PRIVATE "${MAYA_DEVKIT_DIR}/lib/${MAYA_LIB_PREFIX}${library}${MAYA_LIB_SUFFIX}")
endforeach()
add_test(NAME draw_geometry COMMAND draw_geometry_test)

foreach(test_name solver_input relax_solver_reference)
    add_executable(${test_name}_test ${test_name}_test.cpp ../sources/bellColliderSolver.cpp)
    target_compile_features(${test_name}_test PRIVATE cxx_std_11)
    target_include_directories(${test_name}_test PRIVATE ../sources "${MAYA_DEVKIT_DIR}/include")
    target_compile_definitions(${test_name}_test PRIVATE ${MAYA_COMPILE_DEFINITIONS})
    foreach(library Foundation OpenMaya OpenMayaAnim OpenMayaFX OpenMayaRender OpenMayaUI clew)
        target_link_libraries(${test_name}_test PRIVATE "${MAYA_DEVKIT_DIR}/lib/${MAYA_LIB_PREFIX}${library}${MAYA_LIB_SUFFIX}")
    endforeach()
    add_test(NAME ${test_name} COMMAND ${test_name}_test)
endforeach()

add_executable(benchmark_relax_kernel benchmark_relax_kernel.cpp)
target_compile_features(benchmark_relax_kernel PRIVATE cxx_std_11)
target_include_directories(benchmark_relax_kernel PRIVATE ../sources)
